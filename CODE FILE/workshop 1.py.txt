
failed_attempts = 0
limit = 3

while failed_attempts < limit:
    password = input("Enter demo password: ")

    if password == "Student@123":
        print("Login successful!")
        break
    else:
        failed_attempts += 1
        print("Incorrect password.")
        print("Attempts:", failed_attempts)

if failed_attempts == limit:
    print("Security alert: Too many failed attempts!")
